# Independently verify native adapters without service reflection, RPC dependencies or downloads.
cmake_minimum_required(VERSION 3.28)
project(AstraPlatform LANGUAGES CXX)
get_filename_component(ASTRA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ASTRA_OUTPUT "${ASTRA_ROOT}/build")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit downloads" FORCE)
if(WIN32 AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(astra_platform windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(astra_platform linux)
else()
    message(FATAL_ERROR "Platform adapters target Linux or x64 Windows/MSVC")
endif()
find_package(Threads REQUIRED)
include(CTest)
add_library(platform_options INTERFACE)
target_include_directories(platform_options INTERFACE "${ASTRA_ROOT}/common/include")
target_compile_definitions(platform_options INTERFACE ASTRA_PROFILE=0)
target_link_libraries(platform_options INTERFACE Threads::Threads)
if(WIN32)
    target_compile_options(platform_options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc)
    target_compile_definitions(platform_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(platform_options INTERFACE ws2_32 mincore bcrypt)
else()
    target_compile_options(platform_options INTERFACE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)
endif()
add_subdirectory(common "${ASTRA_OUTPUT}/Common/platform")
add_subdirectory(star "${ASTRA_OUTPUT}/Star/platform")

# Source browsing has no compiler detection, try_compile, dependency discovery or downloads.
cmake_minimum_required(VERSION 3.28)
project(Astra LANGUAGES NONE)
if(NOT CMAKE_GENERATOR MATCHES "^Visual Studio " OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "Use tools/build.py solution with an installed x64 Visual Studio generator")
endif()
foreach(input IN ITEMS ASTRA_ROOT ASTRA_OUTPUT ASTRA_PYTHON)
    if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
        message(FATAL_ERROR "Missing absolute generator input: ${input}")
    endif()
endforeach()
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Prepared dependency configuration" FORCE)
# Regeneration belongs to the Python publisher, which rebases the root solution atomically.
set(CMAKE_SUPPRESS_REGENERATION ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT Comet)
foreach(component IN ITEMS Comet Star Pulsar Common Planet)
    string(TOLOWER "${component}" directory)
    if(component STREQUAL "Comet")
        set(directory comet/cpp)
    endif()
    add_subdirectory(project "${ASTRA_OUTPUT}/${component}")
endforeach()

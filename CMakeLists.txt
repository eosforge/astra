# Dependency preparation requires separate explicit authorization; normal configuration only consumes an existing prefix.
# gRPC's export calls find_dependency(ZLIB) again, so consumers must select static libraries too.
find_package(yyjson CONFIG REQUIRED)
# Upstream yyjson installs exported targets without a package-version file; validate the pinned version from the target's actual headers.
get_target_property(star_yyjson_includes yyjson::yyjson INTERFACE_INCLUDE_DIRECTORIES)
find_path(star_yyjson_headers yyjson.h PATHS ${star_yyjson_includes} NO_DEFAULT_PATH REQUIRED)
file(STRINGS "${star_yyjson_headers}/yyjson.h" star_yyjson_version REGEX "^#define YYJSON_VERSION_STRING")
if(NOT star_yyjson_version MATCHES "\"0\\.13\\.0\"")
    message(FATAL_ERROR "Star/Planet requires the prepared yyjson 0.13.0 headers and library")
endif()
set(wire_dir "${CMAKE_CURRENT_SOURCE_DIR}/common/src/generated")
foreach(schema IN ITEMS astra orbit comet pulsar polaris)
    foreach(suffix IN ITEMS pb.cc pb.h grpc.pb.cc grpc.pb.h)
        set(generated_file "${wire_dir}/${schema}.${suffix}")
        if(NOT EXISTS "${generated_file}")
            message(FATAL_ERROR "Missing generated protocol: ${generated_file}. Run the explicit project proto command first.")
        endif()
        file(SIZE "${generated_file}" generated_size)
        if(generated_size EQUAL 0)
            message(FATAL_ERROR "Empty generated protocol: ${generated_file}. Generate real sources; placeholders are not accepted.")
        endif()
    endforeach()
endforeach()
add_library(star_wire STATIC
    "${wire_dir}/astra.pb.cc" "${wire_dir}/orbit.pb.cc" "${wire_dir}/comet.pb.cc" "${wire_dir}/pulsar.pb.cc" "${wire_dir}/polaris.pb.cc"
    "${wire_dir}/astra.grpc.pb.cc" "${wire_dir}/orbit.grpc.pb.cc" "${wire_dir}/comet.grpc.pb.cc" "${wire_dir}/pulsar.grpc.pb.cc" "${wire_dir}/polaris.grpc.pb.cc")

    # Instrument generated messages too, without imposing handwritten warning or reflection rules.

endif()

    target_compile_definitions(star_wire PUBLIC PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT=1)
endif()

if(NOT WIN32)
    # Copy actual compiler runtime files under their loader names; installation must not retain toolchain symlinks or paths.
    foreach(star_runtime_name IN ITEMS libstdc++.so.6 libgcc_s.so.1)
        execute_process(COMMAND "${CMAKE_CXX_COMPILER}" "-print-file-name=${star_runtime_name}"
            OUTPUT_VARIABLE star_runtime_file OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        if(NOT IS_ABSOLUTE "${star_runtime_file}" OR NOT EXISTS "${star_runtime_file}" OR IS_DIRECTORY "${star_runtime_file}")
            message(FATAL_ERROR "The pinned compiler cannot locate ${star_runtime_name} for service installation")
        endif()
        file(REAL_PATH "${star_runtime_file}" star_runtime_file)
        install(FILES "${star_runtime_file}" DESTINATION lib RENAME "${star_runtime_name}")
    endforeach()
    set_target_properties(star pulsar PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH FALSE)
endif()

# Build and package the standalone public C++ SDK with explicit dependency and test inputs.
cmake_minimum_required(VERSION 3.28)
project(Comet VERSION 0.1.0 LANGUAGES CXX)
# The SDK uses C++23 even within a C++26 server build; directory scope preserves the parent project's settings.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASTRA_PROFILE "Compile private SDK performance probes" OFF)

# Standalone SDK configuration does not add the server project or invoke dependency downloads.
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND MSVC_VERSION GREATER_EQUAL 1951)
    # Windows initially supports the standalone SDK, without expanding support to servers or Linux-only probes.
    if(ASTRA_PROFILE OR ASTRA_SANITIZER OR COMET_SANITIZER)
        message(FATAL_ERROR "Comet MSVC builds currently exclude profiling and sanitizers")
    endif()
elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 16.2.0)
    message(FATAL_ERROR "Comet requires Linux/GCC 16.2.0 or Windows/MSVC 19.51 or newer")
endif()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit downloads" FORCE)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(Threads REQUIRED)
find_package(Protobuf 36.1.0 EXACT CONFIG REQUIRED)
find_package(gRPC 1.84.0 EXACT CONFIG REQUIRED)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

get_filename_component(comet_common "${CMAKE_CURRENT_SOURCE_DIR}/../../common" ABSOLUTE)
set(comet_generated "${comet_common}/src/generated")
# Package only public Comet protocols. Generated sources do not inherit handwritten warnings, reflection, or contracts.
add_library(comet_protocol OBJECT "${comet_generated}/comet.pb.cc" "${comet_generated}/comet.grpc.pb.cc")
# Some Protobuf types vary with language mode; generated protocols and the SDK need matching feature requirements.
target_compile_features(comet_protocol PRIVATE cxx_std_23)
target_link_libraries(comet_protocol PRIVATE protobuf::libprotobuf gRPC::grpc++)
target_include_directories(comet_protocol SYSTEM PRIVATE "${comet_generated}")
set_target_properties(comet_protocol PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(comet STATIC src/sampling.cpp src/core.cpp src/reading.cpp src/projection.cpp src/observing.cpp src/selection.cpp src/beaming.cpp src/lifetime.cpp src/subscribing.cpp src/subscription.cpp src/publishing.cpp "${comet_common}/src/types.cpp" "${comet_common}/src/scope.cpp" "${comet_common}/src/profile.cpp" $<TARGET_OBJECTS:comet_protocol>)
add_library(comet::comet ALIAS comet)
# Public headers use expected and move_only_function; propagate the C++23 minimum to source and installed-package consumers.
target_compile_features(comet PUBLIC cxx_std_23)
if(MSVC)
    target_compile_options(comet PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    # Unused legacy TLS overloads in gRPC 1.84 still instantiate deprecated MSVC STL types; SYSTEM headers cannot isolate C4996.
    # The SDK uses InMemoryCertificateProvider; suppress this diagnostic only for this target, never consumers or unrelated warnings.
    target_compile_options(comet PRIVATE /wd4996)
    target_compile_definitions(comet PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(comet PRIVATE ws2_32 mincore)
else()
    target_compile_options(comet PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)
    target_compile_definitions(comet PRIVATE _GLIBCXX_ASSERTIONS=1)
endif()
set_target_properties(comet PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
# Keep private templates consistent within a build; installed public SDK headers neither expose diagnostics types nor force consumers to enable them.
target_compile_definitions(comet PUBLIC "$<BUILD_INTERFACE:$<$<BOOL:${ASTRA_PROFILE}>:ASTRA_PROFILE=1>>")
target_include_directories(comet PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>" PRIVATE src "${comet_common}/include")
target_include_directories(comet SYSTEM PRIVATE "${comet_generated}")
target_link_libraries(comet PRIVATE protobuf::libprotobuf gRPC::grpc++ Threads::Threads)
if(ASTRA_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "SDK coverage requires GCC condition instrumentation")
    endif()
    target_compile_options(comet PRIVATE --coverage -fcondition-coverage -fprofile-update=atomic)
    target_link_options(comet PUBLIC "$<BUILD_INTERFACE:--coverage>")
endif()

# Explicit root sanitizer settings cover both SDK and generated protocols; installed targets retain actual linkage requirements.
set(COMET_SANITIZER "${ASTRA_SANITIZER}" CACHE STRING "Optional address,undefined or thread sanitizer")
if(DEFINED ASTRA_SANITIZER)
    set(COMET_SANITIZER "${ASTRA_SANITIZER}")
endif()
if(COMET_SANITIZER)
    if(ASTRA_PROFILE)
        message(FATAL_ERROR "Performance probes and sanitizers need separate SDK builds")
    endif()
    if(NOT COMET_SANITIZER MATCHES "^(address,undefined|thread)$")
        message(FATAL_ERROR "Comet sanitizer must be empty, address,undefined or thread")
    endif()
    # Propagate instrumentation to consumers and standalone tests. Protobuf adds layout fields under TSan;
    # link-only flags would mix incompatible ABIs, so installed packages must retain compile requirements too.
    target_compile_options(comet PUBLIC "-fsanitize=${COMET_SANITIZER}" -fno-omit-frame-pointer)
    target_compile_options(comet_protocol PRIVATE "-fsanitize=${COMET_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(comet INTERFACE "-fsanitize=${COMET_SANITIZER}")
endif()
if(COMET_SANITIZER MATCHES "address")
    # Match ordinary prebuilt Protobuf 36.1 metadata layout for standalone SDK builds and installed-package consumers.
    # This upstream switch preserves memory instrumentation and consistency checks while preventing a 56/40-byte ClassData ABI mismatch.
    target_compile_definitions(comet PUBLIC PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT=1)
    target_compile_definitions(comet_protocol PRIVATE PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT=1)
endif()

# Embed only explicitly selected public CA data; never generate, download, or install trust material. Runtime CA files take precedence.
set(COMET_CA_FILE "" CACHE FILEPATH "Optional public CA PEM embedded in the SDK")
if(COMET_CA_FILE)
    get_filename_component(comet_ca "${COMET_CA_FILE}" ABSOLUTE)
    if(NOT EXISTS "${comet_ca}" OR IS_DIRECTORY "${comet_ca}" OR comet_ca MATCHES "[\";\n\r]")
        message(FATAL_ERROR "COMET_CA_FILE must name an existing public CA file")
    endif()
    file(SIZE "${comet_ca}" comet_ca_size)
    if(comet_ca_size EQUAL 0 OR comet_ca_size GREATER 1048576)
        message(FATAL_ERROR "Embedded Comet CA must contain 1..1048576 bytes")
    endif()
    file(READ "${comet_ca}" comet_ca_content)
    if(comet_ca_content MATCHES "PRIVATE KEY" OR NOT comet_ca_content MATCHES "-----BEGIN CERTIFICATE-----")
        message(FATAL_ERROR "Embed only public CA certificates, never a private key")
    endif()
    # CMake converts certificates into private byte lists on all platforms without C++26 directives or extra tools.
    file(READ "${comet_ca}" comet_ca_hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," comet_ca_bytes "${comet_ca_hex}")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/comet_ca.inc" "${comet_ca_bytes}\n")
    target_include_directories(comet PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(comet PRIVATE COMET_CA_GENERATED=1)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${comet_ca}")
endif()

# Install only owned headers, static archives, exports, and licenses; consumers find dependencies in their prepared prefixes.
install(TARGETS comet EXPORT CometTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT Comet)
install(DIRECTORY include/comet DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}" COMPONENT Comet)
install(EXPORT CometTargets NAMESPACE comet:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Comet" COMPONENT Comet)
configure_package_config_file(cmake/CometConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/CometConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Comet")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CometConfigVersion.cmake" VERSION "${PROJECT_VERSION}" COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CometConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CometConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Comet" COMPONENT Comet)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" DESTINATION "${CMAKE_INSTALL_DATADIR}/comet" COMPONENT Comet)
